from selenium import webdriver
from bs4 import BeautifulSoup

browser = webdriver.Chrome('./chromedriver')
url ='url 넣기'
browser.get(url)

html = browser.page_source

# html 문서 읽어오기
soup = BeautifulSoup(html, 'html.parser')

test = soup.select('태그')
# 부모태그 > 자식태그
# 결과값은 list

print(test[0].text)
# 0번째의 텍스트만 가져오기

print(test[0]['src']
# 태그 안의 속성값 가져오기
